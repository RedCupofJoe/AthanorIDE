"""Run the exported graph with an isolated worker per agent/harness, or serve MCP."""

from __future__ import annotations

from workflow_scheduler import WorkflowExecutionError, run_workflow as schedule_workflow, workflow_plan

import argparse
import select
import json
import os
from pathlib import Path
import queue
import shutil
import subprocess
import sys
import threading
import time
import uuid

ROOT = Path(__file__).resolve().parent
GRAPH_PATH = ROOT / "agent-workflow.json"
IMAGE = os.environ.get("ATHANOR_IMAGE", "localhost/athanor-agent:latest")
MAX_LINE = 4_000_000
APPROVAL_LOCK = threading.Lock()


def connected(graph, node, port):
    ports = {p["id"] for p in node["inputs"] if p["name"] == port}
    nodes = {n["id"]: n for n in graph["nodes"]}
    return [
        nodes[e["sourceNodeId"]]
        for e in graph["edges"]
        if e["targetNodeId"] == node["id"] and e["targetPortId"] in ports
    ]


def openshell_cpu(value):
    cpus = float(value)
    if cpus <= 0:
        raise ValueError("Unsupported CPU limit")
    return str(int(cpus)) if cpus.is_integer() else str(cpus)


def openshell_policy_filename(sandbox_id):
    # Keep this in step with athanor_api.exporter.sandbox_policy.policy_filename.
    safe = "".join(ch if ch.isalnum() or ch in "._-" else "-" for ch in sandbox_id) or "sandbox"
    return f"openshell-policy-{safe}.yaml"


def oci_user(value):
    if value == "sandbox":
        return "65532:65532"
    return f"{value}:{value}"


def attached_policy(graph, sandbox):
    found = [
        node for node in connected(graph, sandbox, "policy") if node["type"] == "sandbox.policy"
    ]
    if len(found) > 1:
        raise ValueError("Connect one Sandbox Policy to each sandbox runtime.")
    return found[0] if found else None


def openshell_command(sandbox, name, argv, *, image=IMAGE, publish_port=None, policy=None):
    """Launch a worker with the OpenShell CLI. Secret values stay off argv."""
    cfg = sandbox["configuration"]
    if cfg["network"] not in {"none", "bridge"}:
        raise ValueError("Unsupported network mode")
    memory_mb = int(cfg["memoryMb"])
    if memory_mb <= 0:
        raise ValueError("Unsupported memory limit")
    command = [
        "openshell",
        "sandbox",
        "create",
        "--name",
        name,
        "--from",
        image,
        "--cpu",
        openshell_cpu(cfg["cpus"]),
        "--memory",
        f"{memory_mb}Mi",
        "--no-tty",
        "--no-auto-providers",
    ]
    # --forward keeps the sandbox alive and cannot be combined with --no-keep.
    if publish_port is None:
        command.append("--no-keep")
    else:
        command += ["--forward", f"127.0.0.1:{int(publish_port)}"]
    if policy is not None:
        filename = openshell_policy_filename(sandbox["id"])
        path = ROOT / filename
        if not path.is_file():
            raise RuntimeError(f"Missing {filename}. Re-export the workflow.")
        command += ["--policy", str(path)]
    return command + ["--", *argv]


def sandbox_command(sandbox, name, *, image=IMAGE, policy=None):
    runtime = sandbox["type"].split(".", 1)[1]
    if runtime == "firecracker":
        raise RuntimeError(
            "Firecracker requires a provisioned microVM adapter. See agent-deployment.json. No fallback is permitted."
        )
    if runtime == "openshell":
        return openshell_command(
            sandbox,
            name,
            ["python", "-m", "app.agent_worker"],
            image=image,
            policy=policy,
        )
    engine = {
        "podman": "podman",
        "docker": "docker",
        "gvisor": "docker",
        "kata": "podman",
    }.get(runtime)
    if not engine:
        raise ValueError("Unsupported sandbox")
    cfg = sandbox["configuration"]
    policy_cfg = None if policy is None else policy["configuration"]
    network = cfg["network"] if policy_cfg is None else policy_cfg["network"]
    if network not in {"none", "bridge"}:
        raise ValueError("Unsupported network mode")
    command = [engine, "run", "--rm", "-i", "--name", name, "--pull=never"]
    read_only = True if policy_cfg is None else bool(policy_cfg.get("readOnlyRoot", True))
    drop_capabilities = (
        True if policy_cfg is None else bool(policy_cfg.get("dropCapabilities", True))
    )
    no_new_privileges = (
        True if policy_cfg is None else bool(policy_cfg.get("noNewPrivileges", True))
    )
    user = "65532:65532" if policy_cfg is None else oci_user(str(policy_cfg.get("runAsUser", "65532")))
    if read_only:
        command.append("--read-only")
    command += ["--user", user]
    if drop_capabilities:
        command.append("--cap-drop=ALL")
    if no_new_privileges:
        command.append("--security-opt=no-new-privileges")
    command += [
        "--network",
        network,
        "--memory",
        f"{int(cfg['memoryMb'])}m",
        "--memory-swap",
        f"{int(cfg['memoryMb'])}m",
        "--cpus",
        str(cfg["cpus"]),
        "--pids-limit",
        str(int(cfg["pidsLimit"])),
        "--tmpfs",
        "/tmp:rw,noexec,nosuid,size=64m",
    ]
    if runtime == "gvisor":
        command += ["--runtime=runsc"]
    if runtime == "kata":
        command += ["--runtime=kata-runtime"]
    return command


def require_engine(command):
    if not shutil.which(command[0]):
        raise RuntimeError(
            f"Install {command[0]} and the selected runtime. Host execution is disabled."
        )
    if command[0] == "podman":
        check = subprocess.run(
            ["podman", "info", "--format", "{{.Host.Security.Rootless}}"],
            capture_output=True,
            text=True,
            check=True,
            timeout=15,
        )
        if check.stdout.strip() != "true":
            raise RuntimeError("Use rootless Podman for this exported launcher.")


def dependencies(graph, node):
    found = {node["id"]}
    pending = [node]
    while pending:
        current = pending.pop()
        for port in current["inputs"]:
            for source in connected(graph, current, port["name"]):
                if source["id"] not in found:
                    found.add(source["id"])
                    pending.append(source)
    return [n for n in graph["nodes"] if n["id"] in found]


def credential_names(nodes):
    names = {n["configuration"].get("secretName") for n in nodes} - {None, ""}
    for node in nodes:
        names.update(
            name.strip()
            for name in node["configuration"].get("environmentNames", "").split(",")
            if name.strip()
        )
    return sorted(names)


def require_credentials(nodes):
    for name in credential_names(nodes):
        if name not in os.environ:
            raise RuntimeError(f"Set environment variable {name} before running")


def credential_args(nodes):
    require_credentials(nodes)
    # Only names appear in argv; container engines read values from the environment.
    return [arg for name in credential_names(nodes) for arg in ("--env", name)]


def remove_worker(command, name):
    if command[0] == "openshell":
        removal = ["openshell", "sandbox", "delete", name]
    else:
        removal = [command[0], "rm", "--force", name]
    subprocess.run(
        removal,
        stdout=subprocess.DEVNULL,
        stderr=subprocess.DEVNULL,
        timeout=15,
    )


def human_approval(event):
    if not sys.stdin.isatty():
        return False
    deadline = event.get("approvalDeadline", time.monotonic() + 120)
    remaining = max(0, deadline - time.monotonic())
    if not APPROVAL_LOCK.acquire(timeout=remaining):
        return False
    try:
        if time.monotonic() >= deadline:
            return False
        print(f"Approve tool {event['tool']} with arguments {json.dumps(event['arguments'])}? Type yes:", file=sys.stderr, flush=True)
        readable, _, _ = select.select([sys.stdin], [], [], max(0, deadline - time.monotonic()))
        return bool(readable) and sys.stdin.readline().strip() == "yes"
    finally:
        APPROVAL_LOCK.release()


def run_isolated(graph, node, task, approval=human_approval):
    sandbox = connected(graph, node, "sandbox")[0]
    name = "athanor-" + uuid.uuid4().hex
    selected = dependencies(graph, node)
    require_credentials(selected)
    policy = attached_policy(graph, sandbox)
    if sandbox["type"] == "sandbox.openshell":
        names = credential_names(selected)
        if names:
            print(
                "OpenShell does not receive secret values on the command line. "
                "Attach providers that inject: " + ", ".join(names),
                file=sys.stderr,
                flush=True,
            )
        command = sandbox_command(sandbox, name, policy=policy)
    else:
        command = sandbox_command(sandbox, name, policy=policy)
        command += credential_args(selected) + [IMAGE, "python", "-m", "app.agent_worker"]
    require_engine(command)
    process = subprocess.Popen(
        command,
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.DEVNULL,
        text=True,
        bufsize=1,
    )
    events = queue.Queue(maxsize=128)
    deadline = time.monotonic() + float(node["configuration"]["timeoutSeconds"])

    def read_events():
        try:
            while True:
                line = process.stdout.readline(MAX_LINE + 1)
                if not line:
                    break
                if len(line) > MAX_LINE:
                    events.put(
                        {"kind": "error", "message": "Worker output exceeds limit"}
                    )
                    return
                events.put(json.loads(line))
        except Exception:
            events.put({"kind": "error", "message": "Invalid worker response"})
        finally:
            events.put({"kind": "eof"})

    try:
        subset = {
            **graph,
            "nodes": selected,
            "edges": [
                e
                for e in graph["edges"]
                if e["sourceNodeId"] in {n["id"] for n in selected}
                and e["targetNodeId"] in {n["id"] for n in selected}
            ],
        }
        process.stdin.write(
            json.dumps({"graph": subset, "node": node, "task": task}) + "\n"
        )
        process.stdin.flush()
        threading.Thread(target=read_events, daemon=True).start()
        while True:
            remaining = deadline - time.monotonic()
            if remaining <= 0:
                raise TimeoutError("Agent run timeout exceeded")
            try:
                event = events.get(timeout=remaining)
            except queue.Empty:
                raise TimeoutError("Agent run timeout exceeded") from None
            if event["kind"] == "approval":
                # Human input must not suspend the worker's wall-clock timeout.
                def decide(request=event):
                    events.put(
                        {
                            "kind": "approval_decision",
                            "approved": bool(approval({**request, "approvalDeadline": deadline})),
                        }
                    )

                threading.Thread(target=decide, daemon=True).start()
            elif event["kind"] == "approval_decision":
                process.stdin.write(json.dumps({"approved": event["approved"]}) + "\n")
                process.stdin.flush()
            elif event["kind"] == "event":
                print(json.dumps(event), file=sys.stderr)
            elif event["kind"] == "result":
                return event["output"]
            elif event["kind"] in {"error", "eof"}:
                raise WorkflowExecutionError(event.get("errorCode", "execution_failure"))
    finally:
        # Killing the client alone may leave a container running: remove by unique run name.
        try:
            remove_worker(command, name)
        finally:
            if process.poll() is None:
                process.kill()
            process.wait(timeout=5)
            process.stdin.close()
            process.stdout.close()


def run_workflow(graph, task_override=None, runner=run_isolated):
    return schedule_workflow(graph, task_override, runner)


def serve_mcp(graph, node_id):
    node = next(
        n for n in graph["nodes"] if n["id"] == node_id and n["type"] == "mcp.server"
    )
    sandbox = connected(graph, node, "sandbox")[0]
    name = "athanor-mcp-" + uuid.uuid4().hex
    selected = dependencies(graph, node)
    require_credentials(selected)
    policy = attached_policy(graph, sandbox)
    argv = ["python", "-m", "app.mcp_server", node_id]
    if sandbox["type"] == "sandbox.openshell":
        port = (
            int(node["configuration"]["port"])
            if node["configuration"]["transport"] == "streamable-http"
            else None
        )
        command = openshell_command(sandbox, name, argv, publish_port=port, policy=policy)
    else:
        command = sandbox_command(sandbox, name, policy=policy)
        command += credential_args(selected)
        if node["configuration"]["transport"] == "streamable-http":
            port = int(node["configuration"]["port"])
            command += ["--publish", f"127.0.0.1:{port}:{port}"]
        command += [IMAGE, *argv]
    require_engine(command)
    try:
        subprocess.run(command, check=True)
    finally:
        remove_worker(command, name)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("command", choices=["plan", "run", "serve-mcp"])
    parser.add_argument("--task")
    parser.add_argument("--node", help="MCP server node ID")
    args = parser.parse_args()
    graph = json.loads(GRAPH_PATH.read_text())
    if args.command == "plan":
        workflow_plan(graph)
        print(
            json.dumps(
                {
                    "nodes": [
                        {"id": n["id"], "name": n["name"], "type": n["type"]}
                        for n in graph["nodes"]
                    ],
                    "edges": graph["edges"],
                    "workflow": graph["workflow"],
                },
                indent=2,
            )
        )
    elif args.command == "serve-mcp":
        if not args.node:
            parser.error("serve-mcp requires --node")
        serve_mcp(graph, args.node)
    else:
        print(json.dumps(run_workflow(graph, args.task), indent=2))


if __name__ == "__main__":
    main()
