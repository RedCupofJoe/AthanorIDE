"""Generated framework-neutral customer harness HTTP adapter.

Replace run_task with your orchestration entry point. This example is a dependency-free
pass-through; it does not invoke a model or call tools until you implement that behavior.
"""

from __future__ import annotations

import json
import os
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from typing import Any

MAX_REQUEST_BYTES = 1_048_576


def run_task(task: str) -> str:
    """Harness extension point. Replace the pass-through with your agent runtime."""

    return task


class Handler(BaseHTTPRequestHandler):
    def do_POST(self) -> None:
        if self.path != "/run":
            self.send_error(404)
            return
        try:
            size = int(self.headers.get("Content-Length", "0"))
            if size <= 0 or size > MAX_REQUEST_BYTES:
                self.send_error(413)
                return
            payload: Any = json.loads(self.rfile.read(size))
            task = payload.get("input") if isinstance(payload, dict) else None
            if not isinstance(task, str):
                self.send_error(400, "input must be a string")
                return
            result = run_task(task)
            if not isinstance(result, str):
                self.send_error(500, "harness result must be a string")
                return
            body = json.dumps({"output": result}).encode("utf-8")
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
        except (ValueError, json.JSONDecodeError):
            self.send_error(400, "invalid JSON request")

    def log_message(self, format: str, *args: object) -> None:
        # Avoid logging task payloads or credentials from request data.
        del format, args


if __name__ == "__main__":
    host = os.environ.get("ATHANOR_HARNESS_HOST", "127.0.0.1")
    port = int(os.environ.get("ATHANOR_HARNESS_PORT", "8010"))
    ThreadingHTTPServer((host, port), Handler).serve_forever()
